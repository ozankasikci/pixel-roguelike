```cpp
    endInspectorPropertyTable();  // end the property table from above

    ImGui::Separator();
    ImGui::TextUnformatted("Leaf Meshes");
    ImGui::Spacing();

    // Find the current door group's object ID to look up children
    // We need to find the EditorSceneObject whose payload matches this dg
    const EditorSceneObject* doorGroupObj = nullptr;
    for (const auto& obj : document.objects()) {
        if (obj.kind != EditorSceneObjectKind::DoorGroup) continue;
        const auto& payload = std::get<LevelDoorPlacement>(obj.payload);
        if (&payload == &dg) {
            doorGroupObj = &obj;
            break;
        }
    }

    bool foundLeaf = false;
    if (doorGroupObj) {
        for (const auto& child : document.objects()) {
            if (child.kind != EditorSceneObjectKind::Mesh) continue;
            const auto& meshPlacement = std::get<LevelMeshPlacement>(child.payload);
            if (meshPlacement.parentNodeId != dg.nodeId) continue;
            if (!meshPlacement.pivot.has_value()) continue;

            foundLeaf = true;
            // Display leaf info as read-only
            ImGui::PushID(static_cast<int>(child.id));
            ImGui::Indent();
            ImGui::TextColored(ImVec4(0.7f, 0.9f, 0.7f, 1.0f), "%s", meshPlacement.meshId.c_str());
            ImGui::SameLine();
            ImGui::TextDisabled("(leaf)");
            const glm::vec3& p = meshPlacement.pivot.value();
            ImGui::TextDisabled("  Pivot: %.2f, %.2f, %.2f", p.x, p.y, p.z);
            ImGui::Unindent();
            ImGui::PopID();
        }
    }

    if (!foundLeaf) {
        ImGui::TextColored(ImVec4(1.0f, 0.85f, 0.0f, 1.0f),
                           "No leaf meshes found -- this door won't animate.");
    }
```
