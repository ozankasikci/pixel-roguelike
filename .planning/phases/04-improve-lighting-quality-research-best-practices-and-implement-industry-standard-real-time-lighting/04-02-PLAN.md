```glsl
#version 410 core
uniform sampler2D uSrcTex;
uniform vec2 uSrcTexelSize;
in vec2 vTexCoord;
out vec4 fragColor;

void main() {
    vec2 uv = vTexCoord;
    vec2 ts = uSrcTexelSize;
    vec3 a = texture(uSrcTex, uv + vec2(-2.0, -2.0) * ts).rgb;
    vec3 b = texture(uSrcTex, uv + vec2( 0.0, -2.0) * ts).rgb;
    vec3 c = texture(uSrcTex, uv + vec2( 2.0, -2.0) * ts).rgb;
    vec3 d = texture(uSrcTex, uv + vec2(-2.0,  0.0) * ts).rgb;
    vec3 e = texture(uSrcTex, uv + vec2( 0.0,  0.0) * ts).rgb;
    vec3 f = texture(uSrcTex, uv + vec2( 2.0,  0.0) * ts).rgb;
    vec3 g = texture(uSrcTex, uv + vec2(-2.0,  2.0) * ts).rgb;
    vec3 h = texture(uSrcTex, uv + vec2( 0.0,  2.0) * ts).rgb;
    vec3 i = texture(uSrcTex, uv + vec2( 2.0,  2.0) * ts).rgb;
    vec3 j = texture(uSrcTex, uv + vec2(-1.0, -1.0) * ts).rgb;
    vec3 k = texture(uSrcTex, uv + vec2( 1.0, -1.0) * ts).rgb;
    vec3 l = texture(uSrcTex, uv + vec2(-1.0,  1.0) * ts).rgb;
    vec3 m = texture(uSrcTex, uv + vec2( 1.0,  1.0) * ts).rgb;
    vec3 result = e * 0.125 + (a + c + g + i) * 0.03125
                + (b + d + f + h) * 0.0625 + (j + k + l + m) * 0.125;
    fragColor = vec4(result, 1.0);
}
```

5. Create `src/engine/rendering/post/BloomPass.h`:
```cpp
#pragma once
#include <glad/gl.h>
#include <memory>

class Shader;

class BloomPass {
public:
    static constexpr int kMipCount = 5;

    BloomPass();
    ~BloomPass();

    BloomPass(const BloomPass&) = delete;
    BloomPass& operator=(const BloomPass&) = delete;

    void init();
    void resize(int baseWidth, int baseHeight);
    void render(GLuint sceneColorTex, float filterRadius);
    GLuint bloomTexture() const;

private:
    void destroy();

    struct MipLevel {
        GLuint fbo = 0;
        GLuint colorTex = 0;
        int width = 0;
        int height = 0;
    };

    std::unique_ptr<Shader> downsampleShader_;
    std::unique_ptr<Shader> upsampleShader_;
    MipLevel mips_[kMipCount];
    GLuint quadVAO_ = 0;
    GLuint quadVBO_ = 0;
    bool initialized_ = false;
};
```

6. Create `src/engine/rendering/post/BloomPass.cpp`:
   - `init()`: Load downsample and upsample shaders using `Shader::load()`. Create fullscreen quad VAO/VBO (same pattern as CompositePass).
   - `resize(baseWidth, baseHeight)`: Destroy old FBOs if they exist. Create `kMipCount` (5) FBO+texture pairs. Mip 0 is half the base resolution (baseWidth/2, baseHeight/2). Each subsequent mip halves again. Each FBO has a single `GL_RGBA16F` color attachment (HDR). Use `GL_LINEAR` filtering for smooth interpolation between mip levels. Minimum dimension: 1 pixel.
   - `render(sceneColorTex, filterRadius)`:
     a. Downsample pass: For i=0..4, bind mips_[i].fbo, set viewport to mips_[i] dimensions, bind the previous level's texture (i==0 uses sceneColorTex, else mips_[i-1].colorTex), draw fullscreen quad with downsample shader. Set uSrcTexelSize = 1.0/vec2(source width, source height).
     b. Upsample pass: Enable additive blending (`glEnable(GL_BLEND); glBlendFunc(GL_ONE, GL_ONE)`). For i=3..0 (bottom to top), bind mips_[i].fbo, set viewport, bind mips_[i+1].colorTex, draw fullscreen quad with upsample shader using uFilterRadius. Disable blending after.
   - `bloomTexture()`: Return mips_[0].colorTex (the final accumulated bloom at half resolution).
   - `destroy()`: Delete all FBO and texture objects. RAII pattern.
   - Destructor calls destroy().

7. Add `BloomPass.cpp` to the `engine_rendering` CMake target source list in CMakeLists.txt.

Key implementation notes:
- All FBO textures use `GL_RGBA16F` internal format (HDR bloom per Pitfall 3)
- No threshold extraction — HDR downsample naturally emphasizes bright pixels
- filterRadius parameter: use 0.005 as default (passed from PostProcessParams::bloomRadius remapped)
