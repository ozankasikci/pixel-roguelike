**Step 2: Create InspectorUtils.h / InspectorUtils.cpp** — shared helpers used by multiple inspectors:

InspectorUtils.h:
```cpp
#pragma once

#include "editor/core/EditorCommand.h"
#include "editor/scene/EditorSceneDocument.h"

#include <glm/glm.hpp>
#include <string>
#include <vector>

class ContentRegistry;

// Shared transform section — draws position/rotation editors for any scene object.
// Returns true if any value was modified.
bool drawTransformSection(glm::vec3& position, glm::vec3& rotation,
                          EditorSceneDocument& document,
                          EditorCommandStack& commandStack,
                          const EditorSceneDocumentState& beforeState);

// Shared transform section with scale (for meshes, groups).
bool drawTransformSectionWithScale(glm::vec3& position, glm::vec3& rotation, glm::vec3& scale,
                                   EditorSceneDocument& document,
                                   EditorCommandStack& commandStack,
                                   const EditorSceneDocumentState& beforeState);

// Behavior authoring sections — render on_activate/on_enter/on_exit/on_timer behavior lists.
struct BehaviorDeclaration;
void drawBehaviorSections(std::vector<BehaviorDeclaration>& behaviors,
                          EditorSceneDocument& document,
                          EditorCommandStack& commandStack,
                          const EditorSceneDocumentState& beforeState,
                          const ContentRegistry& content,
                          bool showTriggerEvents);

// Interactable section — prompt text, distance, dot threshold.
struct InteractableDeclaration;
void drawInteractableSection(std::optional<InteractableDeclaration>& interactable,
                             EditorSceneDocument& document,
                             EditorCommandStack& commandStack,
                             const EditorSceneDocumentState& beforeState);

// Node ID section — shown for all scene objects with a nodeId field.
void drawNodeIdSection(std::string& nodeId, std::string& parentNodeId,
                       EditorSceneDocument& document,
                       EditorCommandStack& commandStack,
                       const EditorSceneDocumentState& beforeState);
```

InspectorUtils.cpp: Move the following from EditorInspectorPanel.cpp:
