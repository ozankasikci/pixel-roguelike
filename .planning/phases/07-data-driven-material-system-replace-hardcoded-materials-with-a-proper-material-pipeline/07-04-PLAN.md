5. Implement `pollMaterialHotReload()`:
```cpp
void ContentRegistry::pollMaterialHotReload(MaterialTextureLibrary& texLibrary) {
    auto now = std::chrono::steady_clock::now();
    auto elapsed = std::chrono::duration_cast<std::chrono::milliseconds>(now - lastMaterialPoll_);
    if (elapsed.count() < kMaterialPollIntervalMs) return;
    lastMaterialPoll_ = now;

    for (auto& [path, knownTime] : materialFileTimes_) {
        try {
            if (!std::filesystem::exists(path)) continue;
            auto currentTime = std::filesystem::last_write_time(path);
            if (currentTime == knownTime) continue;
            knownTime = currentTime;

            spdlog::info("Hot-reloading material: {}", path);
            auto updated = loadMaterialDefinitionAsset(path);

            // Validate before accepting
            std::string error;
            if (!validateMaterialDefinition(updated, error)) {
                spdlog::error("Hot-reload validation failed for '{}': {}", path, error);
                continue;
            }

            // Update in-memory material
            const std::string oldId = updated.id;
            materials_[updated.id] = std::move(updated);

            // Invalidate texture cache for this material
            texLibrary.reloadMaterial(oldId);
        } catch (const std::exception& e) {
            spdlog::error("Hot-reload error for '{}': {}", path, e.what());
        }
    }
}
```

6. Implement `validateMaterialDefinition()` per D-15:
```cpp
bool ContentRegistry::validateMaterialDefinition(const MaterialDefinition& def, std::string& errorOut) const {
    if (def.id.empty()) {
        errorOut = "Material ID is empty";
        return false;
    }
    if (def.parent.has_value() && !def.parent->empty()) {
        if (!materials_.count(*def.parent)) {
            errorOut = "Parent material '" + *def.parent + "' not found";
            return false;
        }
    }
    if (def.roughnessBias.has_value()) {
        float v = *def.roughnessBias;
        if (v < 0.0f || v > 1.0f) {
            errorOut = "roughness_bias " + std::to_string(v) + " out of range [0, 1]";
            return false;
        }
    }
    if (def.metalness.has_value()) {
        float v = *def.metalness;
        if (v < 0.0f || v > 1.0f) {
            errorOut = "metalness " + std::to_string(v) + " out of range [0, 1]";
            return false;
        }
    }
    return true;
}
```

9. Implement `reloadMaterial()`:
```cpp
void MaterialTextureLibrary::reloadMaterial(const std::string& materialId) {
    // Clear cached entries for this material so next resolve() recomputes them
    resolvedDefinitions_.erase(materialId);
    textureSets_.erase(materialId);
    materials_.erase(materialId);
    spdlog::info("Invalidated texture cache for material: {}", materialId);
}
```

This ensures the next call to `resolve(materialId)` will re-resolve the definition from the updated ContentRegistry, regenerate procedural textures if parameters changed, and rebuild the `RenderMaterialData`. The texture generation is lazy — it happens on next `resolve()` call, not eagerly. Per the anti-pattern from RESEARCH.md, procedural textures only regenerate when their parameters change because the `textureKeyFor()` hash changes.
