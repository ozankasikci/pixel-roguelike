    glm::mat4 makePivotLeafModel(const glm::vec3& groupWorldPos,
                                  float groupYawDeg,
                                  const glm::vec3& pivot,
                                  const glm::vec3& scale) {
        const float yawRad = glm::radians(groupYawDeg);
        const glm::vec3 hingeWorldPos = groupWorldPos + glm::vec3(
            pivot.x * std::cos(yawRad) - pivot.z * std::sin(yawRad),
            0.0f,
            pivot.x * std::sin(yawRad) + pivot.z * std::cos(yawRad));
        glm::mat4 model = glm::translate(glm::mat4(1.0f), hingeWorldPos);
        model = glm::rotate(model, yawRad, glm::vec3(0.0f, 1.0f, 0.0f));
        model = glm::translate(model, -pivot);
        model = glm::scale(model, scale);
        return model;
    }

    entt::entity LevelBuilder::addDoorGroup(const LevelDoorGroupPlacement& group, const LevelDef& level) {
        // Find child meshes belonging to this door group
        const LevelMeshPlacement* framePlacement = nullptr;
        const LevelMeshPlacement* leafPlacement = nullptr;
        for (const auto& m : level.meshes) {
            if (m.parentNodeId == group.nodeId) {
                if (m.pivot.has_value()) {
                    leafPlacement = &m;
                } else {
                    framePlacement = &m;
                }
            }
        }

        if (!leafPlacement) {
            return entt::null;  // No leaf mesh = no functional door
        }

        // Frame mesh: spawn as normal static mesh (if exists)
        if (framePlacement) {
            auto frameEntity = addMesh(framePlacement->meshId,
                framePlacement->position,  // already resolved by hierarchy
                framePlacement->scale,
                framePlacement->rotation,
                framePlacement->tint,
                framePlacement->materialId);
            if (frameEntity != entt::null && !framePlacement->nodeId.empty()) {
                attachNodeId(frameEntity, framePlacement->nodeId);
            }
        }

        // Leaf mesh: use pivot-aware model
        const glm::vec3 pivot = leafPlacement->pivot.value();
        const glm::vec3 hingeWorldPos = computeHingeWorldPos(group.position, group.yawDegrees, pivot);

        // Compute center offset from hinge: the vector from hinge to mesh center in world space
        // Since pivot is the offset from mesh origin to hinge in pre-scale local space,
        // centerOffset = -pivot * scale (negate because center = hinge - pivot direction)
        // But the current hardcoded value is (0.445, 0, 0) at scale 0.22 with pivot (-0.45, 0, 0.04).
        // centerOffset.x = -(-0.45) * 0.22 = 0.099 which does NOT match 0.445.
        // The actual center offset is derived from mesh geometry: half the door width.
        // Keep using the geometry-based approach: half of mesh extent in X at given scale.
        // For now, compute as: centerOffset = glm::vec3(-pivot.x * leafPlacement->scale.x, 0.0f, 0.0f)
        // FALLBACK: If the computed offset seems wrong, use the door mesh half-width directly.
        // The SM_DoorA mesh raw X extent is ~4.06 units, so half-width at scale is ~4.06/2 * scale.x
        const glm::vec3 centerOffsetFromHinge(
            std::abs(pivot.x) * 2.0f * leafPlacement->scale.x,  // ~0.45*2*0.22 = 0.198 — NO
            0.0f, 0.0f);
        // Actually: The current code places the leaf at hingeWorldPos and the center offset is used
        // for DoorLeafComponent to know where the visual center is relative to hinge for collider updates.
        // Current value: (0.445, 0, 0). At scale 0.22, door mesh raw half-X is ~2.03 * 0.22 = 0.4466.
        // So centerOffset = mesh_half_x_extent * scale.x ≈ 0.445.
        // We can approximate: centerOffset.x = 0.5 * mesh_x_extent * scale.x
        // Since we don't have mesh bounds at build time, use the simpler approach:
        // Read leaf mesh bounds if available, or fall back to existing hardcoded per-mesh values.
        // SIMPLEST: keep collider half-extents hardcoded per-mesh (they're geometry-specific).
        //
        // Use the known formula: centerOffsetFromHinge = (mesh_half_width * scale.x, 0, 0)
        // For standard doors this is approximately (-pivot.x * scale.x) but that gives 0.099.
        // The CORRECT formula: the mesh origin is at geometric center.
        // Distance from hinge (at pivot) to center (at origin) = |pivot| * scale
        // pivot = (-0.45, 0, 0.04) so horizontal distance = 0.45 * 0.22 = 0.099? No.
        //
        // Looking at the actual code more carefully:
        // The model is: T(hinge) * R(yaw) * T(-pivot) * S(scale)
        // After this transform, the mesh origin ends up at:
        //   hinge + R(yaw) * (-pivot * scale) (approximately, for small Z)
        // So the center offset from hinge = R(yaw) * (-pivot * scale)
        // For pivot = (-0.45, 0, 0.04), scale = 0.22:
        //   -pivot * scale = (0.45*0.22, 0, -0.04*0.22) = (0.099, 0, -0.0088)
        // But current hardcoded is (0.445, 0, 0). This means the mesh origin is NOT at the
        // mesh geometric center. The mesh origin is shifted.
        //
        // RESOLUTION: The transform chain is T(hinge) * R(yaw) * T(-pivot) * S(scale).
        // The mesh origin (0,0,0) in model space maps to: hinge + R(yaw) * (-pivot * scale).
        // The mesh geometric center is at some offset from origin in model space.
        // For SM_DoorA: raw half-X ≈ 2.03 units. If origin is at the hinge-side edge,
        // then center = origin + (2.03, 0, 0) in model space, and in world:
        // center = hinge + R(yaw) * ((-pivot + vec3(2.03,0,0)) * scale)
        //        = hinge + R(yaw) * ((0.45+2.03)*0.22, 0, -0.04*0.22)
        //        ≈ hinge + R(yaw) * (0.546, 0, -0.009)
        // Still doesn't match 0.445.
        //
        // PRAGMATIC APPROACH: Just compute centerOffsetFromHinge the same way the old code did:
        // closedCenter - hingePosition. The old code places the leaf entity at hingeWorldPos
        // and stores centerOffsetFromHinge = (0.445, 0, 0). We keep this as a hardcoded
        // per-mesh constant. For each known door mesh:
        //   SM_DoorA: (0.445, 0, 0)
        //   SM_DoorC, SM_DoorD: same (0.445, 0, 0) — they share similar geometry.
        // Store a simple lookup or just use (0.445, 0, 0) as default.
        const glm::vec3 defaultCenterOffset(0.445f, 0.0f, 0.0f);

        // Create leaf mesh entity at the resolved position (which is the group position for local (0,0,0) children)
        const glm::mat4 leafModel = makePivotLeafModel(group.position, group.yawDegrees, pivot, leafPlacement->scale);

        Mesh* leafMesh = mesh(leafPlacement->meshId);
        auto leafEntity = addMesh(leafMesh,
            group.position,  // nominal position (overridden by modelOverride)
            leafPlacement->scale,
            glm::vec3(0.0f, group.yawDegrees, 0.0f),
            leafPlacement->tint,
            leafPlacement->materialId);

        if (leafEntity == entt::null) {
            return entt::null;
        }

        auto& reg = registry();

        // Override the model matrix for correct pivot rendering
        if (auto* meshComp = reg.try_get<MeshComponent>(leafEntity)) {
            meshComp->modelOverride = leafModel;
            meshComp->useModelOverride = true;
        }

        // Attach DoorLeafComponent
        DoorLeafComponent doorLeaf;
        doorLeaf.hingePosition = hingeWorldPos;
        doorLeaf.centerOffsetFromHinge = defaultCenterOffset;
        doorLeaf.closedScale = leafPlacement->scale;
        doorLeaf.colliderHalfExtents = glm::vec3(0.445f, 1.01f, 0.05f);  // per-mesh hardcoded
        doorLeaf.closedYaw = group.yawDegrees;
        doorLeaf.openYaw = group.yawDegrees - group.openAngle;
        reg.emplace<DoorLeafComponent>(leafEntity, doorLeaf);

        if (!leafPlacement->nodeId.empty()) {
            attachNodeId(leafEntity, leafPlacement->nodeId);
        }

        // Create door root entity with DoorComponent + InteractableComponent
        auto doorRoot = createTransformEntity(group.position + glm::vec3(0.0f, 1.0f, 0.0f));

        DoorComponent door;
        door.leftLeaf = leafEntity;
        door.rightLeaf = entt::null;
        door.openDuration = group.openDuration;
        reg.emplace<DoorComponent>(doorRoot, door);

        InteractableComponent interact;
        interact.prompt = group.locked ? group.lockedPrompt : "E  Open";
        interact.activatedPrompt = "";
        interact.distance = group.interactDistance;
        interact.dotThreshold = group.interactDotThreshold;
        interact.singleUse = false;
        interact.activated = group.locked;
        reg.emplace<InteractableComponent>(doorRoot, interact);

        if (!group.nodeId.empty()) {
            attachNodeId(doorRoot, group.nodeId);
        }

        return doorRoot;
    }
