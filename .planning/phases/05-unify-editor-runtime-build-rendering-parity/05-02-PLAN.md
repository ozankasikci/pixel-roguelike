```cpp
#pragma once

#include "engine/rendering/SceneRenderPipeline.h"
#include "engine/rendering/core/Framebuffer.h"
#include "game/rendering/MaterialTextureLibrary.h"

#include <vector>

class ContentRegistry;
class EditorPreviewWorld;
class EditorSceneDocument;
struct EnvironmentDefinition;

struct EditorViewportRenderParams {
    // Camera
    glm::mat4 viewMatrix{1.0f};
    glm::mat4 projectionMatrix{1.0f};
    glm::vec3 cameraPosition{0.0f};
    float nearPlane = 0.1f;
    float farPlane = 100.0f;
    // Objects and overlays
    const std::vector<RenderObject>* objects = nullptr;
    const std::vector<RenderLight>* lights = nullptr;
    // Environment
    const EnvironmentDefinition* environment = nullptr;
    // Toggles
    bool shadowsEnabled = true;
    int shadowResolutionIndex = 1;
};

class EditorViewportRenderer {
public:
    void init(const ContentRegistry& content);
    void shutdown();
    void reloadContent(const ContentRegistry& content);

    void render(const EditorViewportRenderParams& params,
                int outputW, int outputH,
                GLuint targetFBO);

    // Access to pipeline's LTC data (for asset preview renderer D-12)
    const LtcData& ltcData() const { return pipeline_.ltcData(); }

private:
    SceneRenderPipeline pipeline_;
    MaterialTextureLibrary materialTextures_;
};
```

Create `src/editor/render/EditorViewportRenderer.cpp`:
```cpp
#include "editor/render/EditorViewportRenderer.h"
#include "game/rendering/EnvironmentDefinition.h"
#include "game/content/ContentRegistry.h"

void EditorViewportRenderer::init(const ContentRegistry& content) {
    pipeline_.init();
    materialTextures_.init(content);
    materialTextures_.prewarmAllMaterialMaps();
}

void EditorViewportRenderer::shutdown() {
    pipeline_.shutdown();
}

void EditorViewportRenderer::reloadContent(const ContentRegistry& content) {
    materialTextures_.init(content);
}

void EditorViewportRenderer::render(const EditorViewportRenderParams& params,
                                     int outputW, int outputH,
                                     GLuint targetFBO) {
    SceneRenderInput input;
    input.objects = params.objects;
    input.lights = params.lights;
    input.viewMatrix = params.viewMatrix;
    input.projectionMatrix = params.projectionMatrix;
    input.cameraPosition = params.cameraPosition;
    input.nearPlane = params.nearPlane;
    input.farPlane = params.farPlane;
    input.postParams = &params.environment->post;
    input.shadowsEnabled = params.shadowsEnabled;
    input.shadowResolutionIndex = params.shadowResolutionIndex;
    // Map EnvironmentDefinition lighting to LightingEnvironment
    input.lightingEnvironment.hemisphereSkyColor = params.environment->lighting.hemisphereSkyColor;
    input.lightingEnvironment.hemisphereGroundColor = params.environment->lighting.hemisphereGroundColor;
    input.lightingEnvironment.hemisphereStrength = params.environment->lighting.hemisphereStrength;
    input.lightingEnvironment.enableDirectionalLights = params.environment->lighting.enableDirectionalLights;
    input.lightingEnvironment.sun = params.environment->lighting.sun;
    input.lightingEnvironment.fill = params.environment->lighting.fill;
    input.lightingEnvironment.enableShadows = params.environment->lighting.enableShadows;
    input.lightingEnvironment.shadowBias = params.environment->lighting.shadowBias;
    input.lightingEnvironment.shadowNormalBias = params.environment->lighting.shadowNormalBias;

    pipeline_.render(input, outputW, outputH, outputW, outputH, targetFBO);
}
```
