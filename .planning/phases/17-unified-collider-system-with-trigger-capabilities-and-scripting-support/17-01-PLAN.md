      ```
      for (const auto& c : data.colliders) {
          ss << "collider ";
          // shape token
          switch (c.shape) {
              case ColliderShape::Box: ss << "box "; break;
              case ColliderShape::Sphere: ss << "sphere "; break;
              case ColliderShape::Cylinder: ss << "cylinder "; break;
              case ColliderShape::Capsule: ss << "capsule "; break;
          }
          // mode token
          switch (c.mode) {
              case ColliderMode::Solid: ss << "solid "; break;
              case ColliderMode::Trigger: ss << "trigger "; break;
              case ColliderMode::SolidAndTrigger: ss << "solidandtrigger "; break;
          }
          // position
          ss << c.position.x << " " << c.position.y << " " << c.position.z << " ";
          // shape-specific params
          if (c.shape == ColliderShape::Box) ss << c.halfExtents.x << " " << c.halfExtents.y << " " << c.halfExtents.z;
          else if (c.shape == ColliderShape::Sphere) ss << c.radius;
          else ss << c.radius << " " << c.halfHeight; // Cylinder + Capsule
          // optional modifiers
          if (c.rotation != glm::vec3(0.0f)) ss << " rotation " << c.rotation.x << " " << c.rotation.y << " " << c.rotation.z;
          if (!c.nodeId.empty()) ss << " node " << c.nodeId;
          if (!c.parentNodeId.empty()) ss << " parent " << c.parentNodeId;
          if (c.fireOnce) ss << " fire_once";
          ss << "\n";
          // behavior sub-lines
          for (const auto& b : c.behaviors) { serializeBehavior(ss, b); }
      }
      ```
