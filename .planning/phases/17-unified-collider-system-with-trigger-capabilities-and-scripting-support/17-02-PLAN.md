    JPH::ShapeRefC makeColliderShape(const ColliderComponent& collider) {
        switch (collider.shape) {
        case ColliderShape::Box:
            return new JPH::BoxShape(toJolt(collider.halfExtents));
        case ColliderShape::Cylinder:
            return new JPH::CylinderShape(collider.halfHeight, collider.radius);
        case ColliderShape::Sphere:

        case ColliderShape::Capsule:

        }
        return nullptr;
    }

    **Step 3: Add SensorContactListener inner class to Impl**

    Inside `PhysicsSystem::Impl` struct, add:
    ```cpp
    class SensorContactListener : public JPH::CharacterContactListener {
    public:
        struct PendingContact {
            entt::entity sensorEntity;
            bool isEnter; // true=enter, false=exit
        };

        void OnContactAdded(const JPH::CharacterVirtual* inCharacter,
                            const JPH::BodyID& inBodyID2,
                            const JPH::SubShapeID& inSubShapeID2,
                            JPH::RVec3Arg inContactPosition,
                            JPH::Vec3Arg inContactNormal,
                            JPH::CharacterContactSettings& ioSettings) override
        {
            auto it = sensorBodyToEntity.find(inBodyID2);
            if (it != sensorBodyToEntity.end()) {
                pending.push_back({it->second, true});
            }
        }

        void OnContactRemoved(const JPH::CharacterVirtual* inCharacter,
                              const JPH::BodyID& inBodyID2,
                              const JPH::SubShapeID& inSubShapeID2) override
        {
            auto it = sensorBodyToEntity.find(inBodyID2);
            if (it != sensorBodyToEntity.end()) {
                pending.push_back({it->second, false});
            }
        }

        std::unordered_map<JPH::BodyID, entt::entity, /* BodyIDHash or std::hash */> sensorBodyToEntity;
        std::vector<PendingContact> pending;
    };
    ```

    For BodyID hashing: JPH::BodyID has a `GetIndexAndSequenceNumber()` method returning uint32_t. Use a simple hash struct:
    ```cpp
    struct BodyIDHash {
        std::size_t operator()(const JPH::BodyID& id) const {
            return std::hash<uint32_t>{}(id.GetIndexAndSequenceNumber());
        }
    };
    ```

    Currently the update loop iterates `registry.view<StaticColliderComponent>()` and creates static bodies for new entities. Change to iterate `registry.view<ColliderComponent>()` and handle three cases:

    Add cleanup for `sensorBodies` and `dualBodies` maps with the same pattern:
    - For `sensorBodies`: remove body from physics system, erase from `sensorListener.sensorBodyToEntity`, erase from map
    - For `dualBodies`: remove both solid and sensor bodies, erase sensor from `sensorListener.sensorBodyToEntity`, erase from map

    **Step 8: Drain pending sensor contacts after ExtendedUpdate**

    After `character->ExtendedUpdate(...)` returns in the update function, add:
    ```cpp
    for (const auto& contact : impl_->sensorListener.pending) {
        auto* collider = registry.try_get<ColliderComponent>(contact.sensorEntity);
        if (!collider || !collider->enabled) continue;
        if (contact.isEnter) {
            if (!collider->playerInside) {
                collider->playerInside = true;
                collider->pendingEnter = true;
            }
            if (collider->fireOnce) collider->enabled = false;
        } else {
            if (collider->playerInside) {
                collider->playerInside = false;
                collider->pendingExit = true;
            }
        }
    }
    impl_->sensorListener.pending.clear();
    ```

    Add cleanup of `sensorBodies` and `dualBodies` maps (remove all bodies from physics system, clear maps and `sensorListener.sensorBodyToEntity`).
