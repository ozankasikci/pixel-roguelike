**2. Implement appendHoverOverlay in EditorScenePreviewRenderer.cpp.**

Add after the `appendSelectionOverlays` function (after line ~265, before `renderShadowPass`):

```cpp
void appendHoverOverlay(std::vector<RenderObject>& objects,
                        const EditorPreviewWorld& world,
                        const MaterialTextureLibrary& materials,
                        std::uint64_t hoveredId,
                        const std::vector<std::uint64_t>& selectedIds) {
    // D-03: no hover on already-selected objects
    if (hoveredId == 0) {
        return;
    }
    for (const auto id : selectedIds) {
        if (id == hoveredId) {
            return;
        }
    }

    Mesh* cube = world.meshLibrary().get("cube");
    const EditorObjectBounds* bounds = world.findObjectBounds(hoveredId);
    if (cube == nullptr || bounds == nullptr || !bounds->valid) {
        return;
    }

    const glm::vec3 center = bounds->center();
    glm::vec3 size = glm::max(bounds->max - bounds->min, glm::vec3(0.12f));
    size *= 1.035f;

    // D-01: cool blue-white at ~60% opacity, line width 2.0
    // D-02: depth-tested only (ignoreDepth = false), single pass (no ghost)
    objects.push_back(RenderObject{
        cube,
        makeModelMatrix(center, size),
        glm::vec3(0.55f, 0.85f, 1.00f),
        materials.resolve("metal_default"),
        true,   // wireframe
        false,  // ignoreDepth = false (depth-tested per D-02)
        true,   // unlit
        2.0f    // lineWidth per D-01
    });
}
```

Note: The tint value `(0.55, 0.85, 1.00)` achieves ~60% effective opacity since the unlit wireframe renderer does not have a separate alpha channel -- the lower-than-1.0 RGB values produce the visual effect of reduced opacity against darker backgrounds.

```cpp
            // Per-frame hover highlight (D-04, D-05)
            std::uint64_t hoveredObjectId = 0;
            const bool suppressHover = gameplayPreviewCaptured
                || placementState.active()
                || editorGizmoIsHot()
                || glfwGetMouseButton(window.handle(), GLFW_MOUSE_BUTTON_RIGHT) == GLFW_PRESS
                || glfwGetMouseButton(window.handle(), GLFW_MOUSE_BUTTON_MIDDLE) == GLFW_PRESS
                || (io.KeyAlt && glfwGetMouseButton(window.handle(), GLFW_MOUSE_BUTTON_LEFT) == GLFW_PRESS);

            if (!suppressHover && renderViewportState.hovered) {
                const EditorRay hoverRay = buildEditorRay(
                    inverseViewProjection,
                    glm::vec2(renderViewportState.origin.x, renderViewportState.origin.y),
                    glm::vec2(renderViewportState.size.x, renderViewportState.size.y),
                    glm::vec2(io.MousePos.x, io.MousePos.y));
                if (const auto hit = pickEditorObject(viewportSelectionHandles, hoverRay)) {
                    hoveredObjectId = hit->objectId;
                }
            }
            appendHoverOverlay(objects, previewWorld, materialTextures, hoveredObjectId, selectedIds);
```

This block:
- Computes hover suppression from all D-04 conditions (play preview captured, placement mode, gizmo drag, right-click, middle-click, alt+left-click)
- Uses `pickEditorObject` (single-hit variant per research recommendation) to find the first object under cursor
- Calls `appendHoverOverlay` which internally checks D-03 (no hover on selected objects)
- Runs every frame unconditionally within the render block (the render block already runs every frame, so no `previewDirty` gating needed per Pitfall 7 in research)
