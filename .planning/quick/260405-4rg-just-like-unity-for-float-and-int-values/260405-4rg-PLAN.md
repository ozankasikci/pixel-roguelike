Click-dragging left/right on the label adjusts the numeric value, matching Unity's UX pattern.

Purpose: Faster value tweaking without needing to precisely click the narrow value widget. Standard UX in Unity, Blender, and other editors.
Output: Modified inspector utilities with label-drag support across all numeric fields.

**In EditorPanelUtils.cpp**, add a thread-local label-drag state manager at file scope (inside the anonymous namespace):

```cpp
struct LabelDragState {
    bool active = false;
    float deltaX = 0.0f;    // Accumulated pixel delta this frame
    float speed = 0.0f;     // Speed of the next DragFloat to consume this
    ImGuiID labelId = 0;    // ID of the label being dragged

    void reset() {
        active = false;
        deltaX = 0.0f;
        speed = 0.0f;
        labelId = 0;
    }
};

thread_local LabelDragState gLabelDrag;
```

**Modify `beginInspectorPropertyLabel`** — for `Scalar` and `Vector` field kinds, replace `ImGui::TextUnformatted(label)` with an invisible button that acts as a drag zone:

```cpp
void beginInspectorPropertyLabel(const char* label, EditorInspectorFieldKind kind) {
    ImGui::TableNextRow();
    ImGui::TableSetColumnIndex(0);
    ImGui::AlignTextToFramePadding();

    const bool isDraggable = (kind == EditorInspectorFieldKind::Scalar
                           || kind == EditorInspectorFieldKind::Vector);

    if (isDraggable) {
        // Render text first to calculate size, then overlay invisible button
        const ImVec2 textSize = ImGui::CalcTextSize(label);
        const ImVec2 cursorPos = ImGui::GetCursorScreenPos();
        const float frameHeight = ImGui::GetFrameHeight();

        ImGui::PushID(label);
        ImGui::PushID("##labeldrag");
        // InvisibleButton sized to fill the label column width
        const float columnWidth = ImGui::GetContentRegionAvail().x;
        ImGui::InvisibleButton("##drag", ImVec2(columnWidth, frameHeight));
        const ImGuiID buttonId = ImGui::GetItemID();

        if (ImGui::IsItemHovered()) {
            ImGui::SetMouseCursor(ImGuiMouseCursor_ResizeEW);
        }

        if (ImGui::IsItemActive() && ImGui::IsMouseDragging(ImGuiMouseButton_Left)) {
            gLabelDrag.active = true;
            gLabelDrag.deltaX = ImGui::GetIO().MouseDelta.x;
            gLabelDrag.labelId = buttonId;
        } else if (gLabelDrag.labelId == buttonId && !ImGui::IsItemActive()) {
            // Released — clear state
            gLabelDrag.reset();
        }

        ImGui::PopID();
        ImGui::PopID();

        // Draw label text over the invisible button
        ImGui::GetWindowDrawList()->AddText(
            ImVec2(cursorPos.x, cursorPos.y + (frameHeight - textSize.y) * 0.5f),
            ImGui::GetColorU32(ImGuiCol_Text),
            label);
    } else {
        ImGui::TextUnformatted(label);
    }

    ImGui::TableSetColumnIndex(1);
    applyInspectorFieldWidth(kind);
    ImGui::PushID(label);
}
```

**Add wrapper functions** that consume label-drag delta. These are drop-in replacements for `ImGui::DragFloat`, `ImGui::DragFloat2`, `ImGui::DragFloat3`, and `ImGui::DragInt`:

```cpp
bool inspectorDragFloat(const char* label, float* v, float speed, float min, float max, const char* format, ImGuiSliderFlags flags) {
    // Apply label-drag delta before rendering widget
    bool changed = false;
    if (gLabelDrag.active && gLabelDrag.deltaX != 0.0f) {
        *v = std::clamp(*v + gLabelDrag.deltaX * speed, min, max);
        changed = true;
    }
    changed |= ImGui::DragFloat(label, v, speed, min, max, format, flags);
    return changed;
}

bool inspectorDragFloat2(const char* label, float v[2], float speed, float min, float max, const char* format, ImGuiSliderFlags flags) {
    bool changed = false;
    if (gLabelDrag.active && gLabelDrag.deltaX != 0.0f) {
        for (int i = 0; i < 2; ++i) {
            v[i] = std::clamp(v[i] + gLabelDrag.deltaX * speed, min, max);
        }
        changed = true;
    }
    changed |= ImGui::DragFloat2(label, v, speed, min, max, format, flags);
    return changed;
}

bool inspectorDragFloat3(const char* label, float v[3], float speed, float min, float max, const char* format, ImGuiSliderFlags flags) {
    bool changed = false;
    if (gLabelDrag.active && gLabelDrag.deltaX != 0.0f) {
        for (int i = 0; i < 3; ++i) {
            v[i] = std::clamp(v[i] + gLabelDrag.deltaX * speed, min, max);
        }
        changed = true;
    }
    changed |= ImGui::DragFloat3(label, v, speed, min, max, format, flags);
    return changed;
}

bool inspectorDragInt(const char* label, int* v, float speed, int min, int max, const char* format, ImGuiSliderFlags flags) {
    bool changed = false;
    if (gLabelDrag.active && gLabelDrag.deltaX != 0.0f) {
        const int delta = static_cast<int>(gLabelDrag.deltaX * speed);
        *v = std::clamp(*v + delta, min, max);
        changed = (delta != 0);
    }
    changed |= ImGui::DragInt(label, v, speed, min, max, format, flags);
    return changed;
}
```

**Important:** The `gLabelDrag` state is set during `beginInspectorPropertyLabel` (column 0) and consumed during the `inspectorDragFloat*` call (column 1) within the same `renderInspectorPropertyRow` call. The state is frame-local since `deltaX` uses `MouseDelta.x` which is per-frame. The label drag state should be reset at the end of each row — add a call at the end of `renderInspectorPropertyRow` or just let it naturally reset when the next label begins (since only one label can be active at a time).
